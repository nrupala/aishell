// Windows Event Log Schema Structuretype WindowsEventLog struct {
	EventID       int    `json:"EventID"`
	ProviderName  string `json:"ProviderName"`
	TimeCreated   string `json:"TimeCreated"` // Local Time ISO-8601 with Offset
	Task          string `json:"Task"`
	UserIntent    string `json:"UserIntent"`
	ScriptPayload string `json:"ScriptPayload"`
	ExecutionResult string `json:"ExecutionResult"`
}
// POSIX System Log Schema Structuretype UnixSyslogMessage struct {
	Timestamp int64  `json:"timestamp"` // Unix Epoch Seconds
	Host      string `json:"host"`
	Process   string `json:"process"`
	Message   string `json:"message"`
	Command   string `json:"command"`
	Status    string `json:"status"`
}
type WatchdogMetrics struct {
	CommandCounts map[string]int `json:"command_counts"`
}
var (

		"sysinfo": "Show system resources, CPU, memory, and OS architecture metrics.",
		"netstat": "List active local network connections, open listening ports, and routing statistics.",
		"diskuse": "Show an aggregate summary of disk storage usage and available partitions.",
	}

	UnixSessionHistory    []UnixSyslogMessage

var DestructiveBlacklist = []string{

	"dd if=", ">/dev/sda", "drop database", "Format-Volume",
	"Remove-Item -Recurse -Force", "shred ",
}
var MutationOperators = []string{
	">", ">>", "Out-File", "Set-Content", "Add-Content", "tee ",

func init() {
	WatchdogData.CommandCounts = make(map[string]int)
	hostOS := runtime.GOOS

	if hostOS == "windows" {
		// Native Windows Path Layout Rules (AppData\Roaming Environment)
		appData := os.Getenv("APPDATA")
		if appData == "" {
			appData = filepath.Join(os.Getenv("USERPROFILE"), "AppData", "Roaming")
		}
		BaseConfigPath = filepath.Join(appData, "ai_shell")
		MetricsFileName = "watchdog_metrics.json"
	} else {
		// Native POSIX/Unix Data Storage Layout Rules (XDG Base Directory Spec)
		homeDir, _ := os.UserHomeDir()
		BaseConfigPath = filepath.Join(homeDir, ".local", "share", "ai_shell")
		MetricsFileName = "watchdog_metrics.json"
	}

	_ = os.MkdirAll(BaseConfigPath, 0755)
	loadWatchdogMetrics()
}
func main() {
	hostOS := runtime.GOOS
	shellEngine, defaultEditor := getOSContext(hostOS)

	printTUIMenu(hostOS, shellEngine)

	reader := bufio.NewReader(os.Stdin)

	for {
		fmt.Printf("\n%s%sAI-Shell>%s ", ColorBold, ColorGreen, ColorReset)
		userInput, err := reader.ReadString('\n')
		if err != nil {
			fmt.Printf("Error reading input: %v\n", err)
			continue
		}

		userInput = strings.TrimSpace(userInput)
		if userInput == "" {
			continue
		}

		if userInput == "exit" || userInput == "quit" {
			promptHistoryFlush(reader, hostOS)
			saveWatchdogMetrics()
			fmt.Println("Exiting AI-Native Shell. Sandbox clean shut-down processed.")
			break
		}

		switch strings.ToLower(userInput) {
		case "help", "-h", "--help":
			printHelpManual()
			continue
		case "aliases", "list-aliases":
			printAliasesTable()
			continue
		}

		if expandedIntent, ok := PreBakedAliases[strings.ToLower(userInput)]; ok {
			fmt.Printf("%s[Alias Intercepted]:%s Expanding macro '%s' -> '%s'\n", ColorBlue, ColorReset, userInput, expandedIntent)
			userInput = expandedIntent
		}

		systemPrompt := buildSystemPrompt(hostOS, defaultEditor, userInput)

		fmt.Printf("\n%s[Thinking Process]%s\n", ColorYellow, ColorReset)
		rawOutput, err := runInference(systemPrompt)
		if err != nil {
			fmt.Printf("%sInference Error:%s %v\n", ColorRed, ColorReset, err)
			continue
		}

		commandPayload := extractCommandBlock(rawOutput)
		commandPayload = strings.TrimSpace(commandPayload)

		if commandPayload == "" {
			fmt.Printf("%s[Security/Parsing Guard]: No valid or safe command block was extracted.%s\n", ColorRed, ColorReset)
			continue
		}

		isSafe, violatesBlacklist := verifySandboxSafety(commandPayload)
		if !isSafe {
			if violatesBlacklist {
				fmt.Printf("%s[CRITICAL SECURITY BLOCK]: Command aborted. Destructive pattern detected.%s\n", ColorRed, ColorReset)
			} else {
				fmt.Printf("%s[Sandbox Intercept]: Active write transaction caught. Redirecting to native Editor...%s\n", ColorYellow, ColorReset)
				commandPayload = formatEditorFallback(hostOS, defaultEditor, commandPayload)
				isSafe = true
			}
		}

		if !isSafe {
			recordPlatformLog(hostOS, userInput, commandPayload, "BLOCKED_SECURITY")
			continue
		}

		fmt.Printf("\n%sProposed Command Line Payload:%s\n", ColorCyan, ColorReset)
		fmt.Printf("-----------------------------------------------------------------\n")
		fmt.Printf("%s%s%s\n", ColorYellow, commandPayload, ColorReset)
		fmt.Printf("-----------------------------------------------------------------\n")
		fmt.Printf("Execute command on host system? [y/N]: ")

		confirmInput, _ := reader.ReadString('\n')
		confirmInput = strings.ToLower(strings.TrimSpace(confirmInput))

		if confirmInput == "y" || confirmInput == "yes" {
			fmt.Printf("%s[Executing Subprocess]%s\n", ColorGreen, ColorReset)
			executeCommand(hostOS, commandPayload)
			recordPlatformLog(hostOS, userInput, commandPayload, "EXECUTED")
			processWatchdog(commandPayload)
		} else {
			fmt.Printf("%s[Execution Aborted by User]%s\n", ColorRed, ColorReset)
			recordPlatformLog(hostOS, userInput, commandPayload, "USER_DECLINED")
		}
	}
}
func verifySandboxSafety(command string) (bool, bool) {
	lowerCmd := strings.ToLower(command)
	for _, restricted := range DestructiveBlacklist {
		if strings.Contains(lowerCmd, strings.ToLower(restricted)) {
			return false, true
		}
	}
	for _, op := range MutationOperators {
		if strings.Contains(command, op) {
			return false, false
		}
	}
	return true, false
}
func formatEditorFallback(hostOS, editor, originalCommand string) string {
	parts := strings.Fields(originalCommand)
	targetFile := "sandbox_output.txt"
	if len(parts) > 0 {
		targetFile = parts[len(parts)-1]
		targetFile = strings.Trim(targetFile, `"' `)
	}

	if hostOS == "windows" {
		return fmt.Sprintf("Start-Process %s '%s'", editor, targetFile)
	}
	return fmt.Sprintf("%s %s", editor, targetFile)
}
func processWatchdog(command string) {
	cleanedCmd := strings.TrimSpace(strings.Split(command, "\n")[0])
	if len(cleanedCmd) > 30 {
		cleanedCmd = cleanedCmd[:30] + "..."
	}

	WatchdogData.CommandCounts[cleanedCmd]++
	if WatchdogData.CommandCounts[cleanedCmd] == 3 {
		fmt.Printf("\n%s💡 [Watchdog Notification]: You have run this command sequence 3 times.%s\n", ColorMagenta, ColorReset)
		fmt.Printf("   Pattern: '%s'\n", cleanedCmd)
		fmt.Println("   Consider saving this query inside a permanent macro shortcut wrapper.")
	}
}
// Implements strict platform-specific architectural event logger routingfunc recordPlatformLog(hostOS, intent, command, status string) {
	if hostOS == "windows" {
		event := WindowsEventLog{
			EventID:         4624, // Arbitrary standard custom Security Execution Audit Code
			ProviderName:    "AIShellEngine",
			TimeCreated:     time.Now().Format("2006-01-02T15:04:05.0000000-07:00"), // Local ISO
			Task:            "AI_Intent_Translation_Gate",
			UserIntent:      intent,
			ScriptPayload:   command,
			ExecutionResult: status,
		}
		WindowsSessionHistory = append(WindowsSessionHistory, event)
	} else {
		hostname, _ := os.Hostname()
		syslogMsg := UnixSyslogMessage{
			Timestamp: time.Now().Unix(), // POSIX Epoch time
			Host:      hostname,
			Process:   "ai_shell",
			Message:   fmt.Sprintf("Intent process complete. Input value: '%s'", intent),
			Command:   command,
			Status:    status,
		}
		UnixSessionHistory = append(UnixSessionHistory, syslogMsg)
	}
}
// Implements platform behavior rules during session persistence save tasksfunc promptHistoryFlush(reader *bufio.Reader, hostOS string) {
	var recordCount int
	if hostOS == "windows" {
		recordCount = len(WindowsSessionHistory)
	} else {
		recordCount = len(UnixSessionHistory)
	}

	if recordCount == 0 {
		return
	}

	// Platform Behavioral Verbiage Segregation
	promptStr := fmt.Sprintf("\nWould you like to archive transaction session event records (%d logs)? [y/N]: ", recordCount)
	if hostOS == "windows" {
		promptStr = fmt.Sprintf("\nExport structural AppData Windows Application Event Logs (%d entries)? [y/N]: ", recordCount)
	}

	fmt.Print(promptStr)
	choice, _ := reader.ReadString('\n')
	choice = strings.ToLower(strings.TrimSpace(choice))

	if choice == "y" || choice == "yes" {
		var data []byte
		var outFileName string
		historyFolder := filepath.Join(BaseConfigPath, "history")
		_ = os.MkdirAll(historyFolder, 0755)

		if hostOS == "windows" {
			outFileName = fmt.Sprintf("AppEventLog_%s.json", time.Now().Format("20060102_150405"))
			data, _ = json.MarshalIndent(WindowsSessionHistory, "", "  ")
		} else {
			outFileName = fmt.Sprintf("syslog_%d.json", time.Now().Unix())
			data, _ = json.MarshalIndent(UnixSessionHistory, "", "  ")
		}

		fullPath := filepath.Join(historyFolder, outFileName)
		err := os.WriteFile(fullPath, data, 0644)
		if err == nil {
			fmt.Printf("%s[Log Transaction Written Safely]:%s Target destination -> %s\n", ColorGreen, ColorReset, fullPath)
		} else {
			fmt.Printf("%s[IO Flush Fault]: Failed writing operational event trail.%s\n", ColorRed, ColorReset)
		}
	}
}
func loadWatchdogMetrics() {
	target := filepath.Join(BaseConfigPath, MetricsFileName)
	if _, err := os.Stat(target); err == nil {
		data, err := os.ReadFile(target)
		if err == nil {
			_ = json.Unmarshal(data, &WatchdogData)
		}

}
}

}
}

func extractCommandBlock(output string) string {
fences := []string{"cmd", "powershell", "bash", "zsh", "shell", ""}
for _, fence := range fences {
if idx := strings.Index(output, fence); idx != -1 {
remaining := output[idx+len(fence):]
endIdx := strings.Index(remaining, "```")
if endIdx != -1 {
return remaining[:endIdx]
}
}
}
return ""
}
func executeCommand(osPlatform, commandPayload string) {
var cmd *exec.Cmd
switch osPlatform {
case "windows":
cmd = exec.Command("powershell.exe", "-Command", commandPayload)
case "linux":
cmd = exec.Command("/bin/bash", "-c", commandPayload)
case "darwin":
cmd = exec.Command("/bin/zsh", "-c", commandPayload)
default:
cmd = exec.Command("/bin/sh", "-c", commandPayload)
}
cmd.Stdout = os.Stdout
cmd.Stderr = os.Stderr
cmd.Stdin = os.Stdin
if err := cmd.Run(); err != nil {
fmt.Printf("\n%sRuntime Execution Error:%s %v\n", ColorRed, ColorReset, err)
}
}
